    90→function toDatetimeLocal(d: Date): string {
    91→  const pad = (n: number) => String(n).padStart(2, "0");
    92→  return `${d.getFullYear()}-${pad(d.getMonth() + 1)}-${pad(d.getDate())}T${pad(
    93→    d.getHours()
    94→  )}:${pad(d.getMinutes())}`;
    95→}

    97→// Deterministic pseudo distance estimate derived from the two location names.
    98→function estimateDistanceKm(a: string, b: string): number {
    99→  const s = `${a.trim().toLowerCase()}->${b.trim().toLowerCase()}`;
   100→  let h = 0;
   101→  for (let i = 0; i < s.length; i++) {
   102→    h = (h * 31 + s.charCodeAt(i)) >>> 0;
   103→  }
   104→  return Math.round((2 + (h % 4300) / 100) * 10) / 10; // 2.0 - 44.9 km
   105→}

   180→  const drop = watch("dropLocation");

   185→    const d = getValues("dropLocation");

   187→      setDistanceKm(estimateDistanceKm(p, d));

   200→    if (isNaN(n)) return;

   203→    const clamped = Math.round(Math.max(distanceMin, Math.min(distanceMax, n)) * 1000) / 1000;

   211→        method: "POST",
   212→        headers: { "Content-Type": "application/json" },
   213→        body: JSON.stringify({
   214→          passengerName: values.passengerName,
   215→          phone: values.phone,
   216→          email: values.email || null,
   217→          pickupLocation: values.pickupLocation,
   218→          dropLocation: values.dropLocation,
   219→          tripType,
   220→          // Pickup time only applies to one-way trips

   225→          cabType,
   226→          passengers: Number(values.passengers),
   227→          distanceKm,
   228→          waitingMinutes: tripType === "ROUND_TRIP" ? 0 : waitingMinutes,

   238→        title: "Ride confirmed 🚕",
   239→        description: `Booking ${data.booking.bookingRef} is confirmed. Your cab is on its way!`,
   240→      });

   242→        passengerName: "",
   243→        phone: "",
   244→        email: "",
   245→        pickupLocation: "",
   246→        dropLocation: "",

   248→        passengers: "1",
   249→        tripType,
   250→      });

   252→      setDistanceTouched(false);
   253→      setDistanceKm(tripType === "ROUND_TRIP" ? ROUND_TRIP_MIN_KM : 8);

   257→        title: "Booking failed",
   258→        description:
   259→          err instanceof Error ? err.message : "Something went wrong. Please try again.",

   264→    }

   278→            {/* Passenger details */}
   279→            <div>
   280→              <h3 className="mb-3 text-sm font-semibold tracking-wide text-muted-foreground uppercase">
   281→                Passenger details
   282→              </h3>
   283→              <div className="grid gap-4 sm:grid-cols-2">

   285→                  <Label htmlFor="passengerName">Full name *</Label>

   287→                    <User className="absolute top-1/2 left-3 size-4 -translate-y-1/2 text-muted-foreground" />
   288→                    <Input
   289→                      id="passengerName"
   290→                      placeholder="e.g. Priya Sharma"
   291→                      className="pl-9"
   292→                      aria-invalid={!!errors.passengerName}
   293→                      {...register("passengerName")}
   294→                    />
   295→                  </div>
   296→                  {errors.passengerName && (
   297→                    <p className="text-xs text-destructive">{errors.passengerName.message}</p>
   298→                  )}
   299→                </div>
   300→                <div className="space-y-2">
   301→                  <Label htmlFor="phone">Phone number *</Label>

   303→                    <Phone className="absolute top-1/2 left-3 size-4 -translate-y-1/2 text-muted-foreground" />
   304→                    <Input
   305→                      id="phone"
   306→                      type="tel"
   307→                      placeholder="+91 98765 43210"
   308→                      className="pl-9"
   309→                      aria-invalid={!!errors.phone}
   310→                      {...register("phone")}
   311→                    />
   312→                  </div>
   313→                  {errors.phone && (
   314→                    <p className="text-xs text-destructive">{errors.phone.message}</p>
   315→                  )}
   316→                </div>
   317→                <div className="space-y-2 sm:col-span-2">
   318→                  <Label htmlFor="email">
   319→                    Email <span className="text-muted-foreground">(optional)</span>
