    67→    67→function toDatetimeLocal(d: Date): string {
    68→    68→  const pad = (n: number) => String(n).padStart(2, "0");
    69→    69→  return `${d.getFullYear()}-${pad(d.getMonth() + 1)}-${pad(d.getDate())}T${pad(
    70→    70→    d.getHours()
    71→    71→  )}:${pad(d.getMinutes())}`;
    72→    72→}

    74→    74→// Deterministic pseudo distance estimate derived from the two location names.
    75→    75→function estimateDistanceKm(a: string, b: string): number {
    76→    76→  const s = `${a.trim().toLowerCase()}->${b.trim().toLowerCase()}`;
    77→    77→  let h = 0;
    78→    78→  for (let i = 0; i < s.length; i++) {
    79→    79→    h = (h * 31 + s.charCodeAt(i)) >>> 0;
    80→    80→  }
    81→    81→  return Math.round((2 + (h % 4300) / 100) * 10) / 10; // 2.0 - 44.9 km
    82→    82→}
